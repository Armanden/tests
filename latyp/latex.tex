\documentclass{beamer}

\title{LaTeX vs Typst}
\author{Comparison Overview}
\date{\today}

\begin{document}

\frame{\titlepage}

\begin{frame}{What are they?}
\begin{itemize}
  \item LaTeX: Mature typesetting system based on TeX
  \item Typst: Modern markup-based typesetting system
\end{itemize}
\end{frame}

\begin{frame}{Key Differences}
\begin{itemize}
  \item LaTeX is decades old; Typst is new
  \item LaTeX is complex; Typst is simple
  \item LaTeX is slow to compile; Typst is fast
  \item LaTeX has massive ecosystem; Typst is growing
\end{itemize}
\end{frame}

\begin{frame}{LaTeX Pros}
\begin{itemize}
  \item Extremely mature and stable
  \item Huge academic adoption
  \item Powerful packages (TikZ, BibTeX)
  \item Best for complex scientific documents
\end{itemize}
\end{frame}

\begin{frame}{LaTeX Cons}
\begin{itemize}
  \item Steep learning curve
  \item Slow compilation
  \item Cryptic errors
  \item Fragmented syntax
\end{itemize}
\end{frame}

\begin{frame}{Typst Pros}
\begin{itemize}
  \item Very fast compilation
  \item Clean, modern syntax
  \item Easier to learn
  \item Consistent built-in system
\end{itemize}
\end{frame}

\begin{frame}{Typst Cons}
\begin{itemize}
  \item Smaller ecosystem
  \item Less academic adoption
  \item Still evolving
\end{itemize}
\end{frame}

\begin{frame}{Conclusion}
\begin{itemize}
  \item Use LaTeX for academic rigor and legacy workflows
  \item Use Typst for speed, simplicity, and modern workflows
\end{itemize}
\end{frame}

\end{document}
